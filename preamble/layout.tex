\renewcommand*{\pagemark}{%
    \tikz[baseline=(mp-page-number.base)]{%
        \node[
            name=mp-page-number,
            circle,
            fill=MPRibbonFrame,
            draw=MPRibbonFrame,
            line width=0.8pt,
            minimum size=8.5mm,
            inner sep=1pt,
            outer sep=0pt
        ] {%
            {\usekomafont{pagenumber}\rl{\thepage}}%
        };
    }%
}

% ==========================================
% PAGE NUMBER VISIBILITY
% ==========================================

\newif\ifMPPageNumbersEnabled
\MPPageNumbersEnabledfalse

% ==========================================
% PAGE NUMBER BADGES
% ==========================================

% Even pages: physical bottom-left corner
\newcommand*{\MPPageNumberLeft}{%
\ifMPPageNumbersEnabled
    \begin{tikzpicture}[remember picture,overlay]
        \node[
            anchor=south west,
            draw=MPRibbonFrame,
            fill=none,
            line width=1.15pt,
            line cap=round,
            line join=round,
            rounded corners=10pt,
            minimum size=8.5mm,
            inner sep=2pt
        ]
        at ([xshift=12mm,yshift=10mm]
            current page.south west)
        {%
            {\usekomafont{pagenumber}
                \rl{\thepage}}%
        };
    \end{tikzpicture}%
\fi
}

% Odd pages: physical bottom-right corner
\newcommand*{\MPPageNumberRight}{%
\ifMPPageNumbersEnabled
    \begin{tikzpicture}[remember picture,overlay]
        \node[
            anchor=south east,
            draw=MPRibbonFrame,
            fill=none,
            line width=1.15pt,
            line cap=round,
            line join=round,
            rounded corners=10pt,
            minimum size=8.5mm,
            inner sep=2pt
        ]
        at ([xshift=-12mm,yshift=10mm]
            current page.south east)
        {%
            {\usekomafont{pagenumber}
                \rl{\thepage}}%
        };
    \end{tikzpicture}%
\fi
}
